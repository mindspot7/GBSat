# Host test: renders the same synthetic Saturn scene with the software and the Vulkan VDP renderers and compares them.
# Built and run by .github/workflows/vulkan-dev.yml on a software Vulkan driver (lavapipe).
cmake_minimum_required(VERSION 3.28)
project(ymir_vk_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(Ymir_DEV_BUILD OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_DEVLOG OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." ymir-build)
set(CEREAL_INSTALL OFF CACHE BOOL "" FORCE)
set(SKIP_PERFORMANCE_COMPARISON ON CACHE BOOL "" FORCE)
set(BUILD_SANDBOX OFF CACHE BOOL "" FORCE)
set(BUILD_DOC OFF CACHE BOOL "" FORCE)
set(JUST_INSTALL_CEREAL ON CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(cereal GIT_REPOSITORY https://github.com/USCiLab/cereal.git GIT_TAG v1.3.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(cereal)
add_executable(vdp_compare vdp_compare.cpp state_loader.cpp)
target_include_directories(vdp_compare PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../android/app/src/main/cpp")
target_link_libraries(vdp_compare PRIVATE ymir::ymir-core cereal::cereal)
add_executable(sw_fastpath_test sw_fastpath_test.cpp)
target_link_libraries(sw_fastpath_test PRIVATE ymir::ymir-core)
